cmake_minimum_required(VERSION 3.25)
project(ElderShellPreview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
get_filename_component(REPO_ROOT "../.." ABSOLUTE)
find_package(imgui CONFIG REQUIRED)
# Reuse the production identity without loading the game/plugin build.
file(READ "${REPO_ROOT}/mod.json" identity)
string(JSON MOD_DISPLAY_NAME_CPP GET "${identity}" displayName)
string(JSON MOD_SHORT_NAME GET "${identity}" shortName)
string(JSON MOD_BINARY_NAME GET "${identity}" binaryName)
string(JSON MOD_SCRIPT_PREFIX GET "${identity}" scriptPrefix)
string(JSON MOD_AUTHOR_CPP GET "${identity}" author)
string(JSON MOD_VERSION GET "${identity}" version)
string(REPLACE "." ";" version_parts "${MOD_VERSION}")
list(GET version_parts 0 PROJECT_VERSION_MAJOR)
list(GET version_parts 1 PROJECT_VERSION_MINOR)
list(GET version_parts 2 PROJECT_VERSION_PATCH)
configure_file("${REPO_ROOT}/cmake/ModInfo.h.in" generated/ModInfo.h @ONLY)
include(Theme.cmake)
add_library(eldershell-preview-ui SHARED UIModule.cpp Simulation.cpp Style.cpp
  "${REPO_ROOT}/src/native/ui/Menu.cpp" "${REPO_ROOT}/src/native/ui/Variables.cpp")
add_custom_command(TARGET eldershell-preview-ui POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E touch "${CMAKE_CURRENT_BINARY_DIR}/ui-ready")
add_executable(eldershell-ui-preview WIN32 Main.cpp ReloadableUI.cpp)
set_target_properties(eldershell-ui-preview PROPERTIES OUTPUT_NAME eldershell-ui-preview-host)
add_dependencies(eldershell-ui-preview eldershell-preview-ui)
foreach(target eldershell-preview-ui eldershell-ui-preview)
  target_include_directories(${target} PRIVATE include "${REPO_ROOT}/src/native" "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_precompile_headers(${target} PRIVATE <algorithm> <array> <atomic> <format> <memory> <mutex> <string> <vector>)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  target_link_libraries(${target} PRIVATE d3d11 dxgi)
endforeach()
target_link_libraries(eldershell-preview-ui PRIVATE imgui::imgui)
