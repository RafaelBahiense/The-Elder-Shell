cmake_minimum_required(VERSION 3.25)
include(cmake/ModIdentity.cmake)
project(${MOD_BINARY_NAME} VERSION ${MOD_VERSION} LANGUAGES CXX)
configure_file(cmake/ModInfo.h.in generated/ModInfo.h @ONLY)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(BUILD_PLUGIN "Build the SKSE plugin" ON)
include(cmake/CapricaEmbedded.cmake)
file(GLOB_RECURSE NATIVE_SOURCES CONFIGURE_DEPENDS src/native/*.cpp)
file(GLOB_RECURSE CONSOLE_TEST_SOURCES CONFIGURE_DEPENDS tests/native/*.cpp)
list(FILTER CONSOLE_TEST_SOURCES EXCLUDE REGEX "/compiler/")
file(GLOB_RECURSE COMPILER_TEST_SOURCES CONFIGURE_DEPENDS tests/native/compiler/*.cpp)

if(BUILD_PLUGIN)
  if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The plugin requires Windows and MSVC")
  endif()
  set(COMMONLIB_SSE_FOLDER "$ENV{COMMONLIB_SSE_FOLDER}" CACHE PATH "CommonLibSSE-NG source")
  if(NOT EXISTS "${COMMONLIB_SSE_FOLDER}/CMakeLists.txt")
    message(FATAL_ERROR "Set COMMONLIB_SSE_FOLDER to CommonLibSSE-NG source")
  endif()
  set(BUILD_TESTS OFF CACHE BOOL "CommonLib tests" FORCE)
  set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_VR ON CACHE BOOL "" FORCE)
  # Keep local checkout paths out of embedded diagnostics and source locations.
  file(TO_NATIVE_PATH "${PROJECT_SOURCE_DIR}" SOURCE_PATHMAP_ROOT)
  file(TO_NATIVE_PATH "${COMMONLIB_SSE_FOLDER}" COMMONLIB_PATHMAP_ROOT)
  add_compile_options(/experimental:deterministic "/pathmap:${SOURCE_PATHMAP_ROOT}=." "/pathmap:${COMMONLIB_PATHMAP_ROOT}=CommonLibSSE-NG")
  add_subdirectory("${COMMONLIB_SSE_FOLDER}" "dependencies/CommonLibSSE-NG")
  find_path(SKSE_MCP_INCLUDE_DIRS "SKSEMCP/SKSEMenuFramework.hpp" REQUIRED)
  add_library(mod SHARED ${NATIVE_SOURCES})
  set_target_properties(mod PROPERTIES OUTPUT_NAME "${MOD_BINARY_NAME}")
  target_link_libraries(mod PRIVATE CommonLibSSE::CommonLibSSE bcrypt)
  target_link_libraries(mod PRIVATE caprica_embedded)
  target_include_directories(mod PRIVATE src/native "${SKSE_MCP_INCLUDE_DIRS}" "${PROJECT_BINARY_DIR}/generated")
  target_precompile_headers(mod PRIVATE src/native/PCH.h)
  target_compile_definitions(mod PRIVATE __CMAKE_COMMONLIBSSE_PLUGIN=1 WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(mod PRIVATE /W4 /permissive- /Zc:preprocessor /EHsc /utf-8)
  target_link_options(mod PRIVATE "/PDBALTPATH:%_PDB%")
  add_library(native_analysis OBJECT EXCLUDE_FROM_ALL ${NATIVE_SOURCES})
  target_link_libraries(native_analysis PRIVATE CommonLibSSE::CommonLibSSE)
  target_link_libraries(native_analysis PRIVATE caprica_embedded)
  target_include_directories(native_analysis PRIVATE src/native "${SKSE_MCP_INCLUDE_DIRS}" "${PROJECT_BINARY_DIR}/generated")
  target_compile_definitions(native_analysis PRIVATE __CMAKE_COMMONLIBSSE_PLUGIN=1 WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(native_analysis PRIVATE /analyze /analyze:external- /WX /W4 /permissive- /Zc:preprocessor /EHsc /utf-8 "/FI${PROJECT_SOURCE_DIR}/src/native/PCH.h")
  get_target_property(COMMONLIB_INCLUDES CommonLibSSE INTERFACE_INCLUDE_DIRECTORIES)
  target_include_directories(native_analysis SYSTEM PRIVATE ${COMMONLIB_INCLUDES})
  install(TARGETS mod RUNTIME DESTINATION SKSE/Plugins)
endif()

enable_testing()
find_package(doctest CONFIG REQUIRED)
add_executable(console_tests ${CONSOLE_TEST_SOURCES})
target_include_directories(console_tests PRIVATE src/native "${PROJECT_BINARY_DIR}/generated")
target_link_libraries(console_tests PRIVATE doctest::doctest)
if(MSVC)
  target_compile_options(console_tests PRIVATE /W4 /WX /utf-8)
endif()
add_test(NAME console_tests COMMAND console_tests)
if(WIN32)
  add_executable(compiler_tests ${COMPILER_TEST_SOURCES} src/native/compiler/Compiler.cpp src/native/compiler/EmbeddedCompiler.cpp)
  target_include_directories(compiler_tests PRIVATE src/native "${PROJECT_BINARY_DIR}/generated")
  target_link_libraries(compiler_tests PRIVATE doctest::doctest bcrypt)
  target_link_libraries(compiler_tests PRIVATE caprica_embedded psapi)
  target_compile_definitions(compiler_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(compiler_tests PRIVATE /W4 /WX /utf-8)
  add_test(NAME compiler_tests COMMAND compiler_tests)
endif()
