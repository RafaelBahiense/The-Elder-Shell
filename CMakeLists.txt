cmake_minimum_required(VERSION 3.25)
project(PapyrusConsole VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(BUILD_PLUGIN "Build the SKSE plugin" ON)

if(BUILD_PLUGIN)
  if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The plugin requires Windows and MSVC")
  endif()
  set(COMMONLIB_SSE_FOLDER "$ENV{COMMONLIB_SSE_FOLDER}" CACHE PATH "CommonLibSSE-NG source")
  if(NOT EXISTS "${COMMONLIB_SSE_FOLDER}/CMakeLists.txt")
    message(FATAL_ERROR "Set COMMONLIB_SSE_FOLDER to CommonLibSSE-NG source")
  endif()
  set(BUILD_TESTS OFF CACHE BOOL "CommonLib tests" FORCE)
  set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_VR ON CACHE BOOL "" FORCE)
  add_subdirectory("${COMMONLIB_SSE_FOLDER}" "dependencies/CommonLibSSE-NG")
  find_path(SKSE_MCP_INCLUDE_DIRS "SKSEMCP/SKSEMenuFramework.hpp" REQUIRED)
  add_library(PapyrusConsole SHARED src/plugin.cpp src/Menu.cpp src/VM.cpp src/GetterDispatch.cpp src/Compiler.cpp src/VMImports.cpp)
  target_link_libraries(PapyrusConsole PRIVATE CommonLibSSE::CommonLibSSE bcrypt)
  target_include_directories(PapyrusConsole PRIVATE src "${SKSE_MCP_INCLUDE_DIRS}")
  target_precompile_headers(PapyrusConsole PRIVATE src/PCH.h)
  target_compile_definitions(PapyrusConsole PRIVATE __CMAKE_COMMONLIBSSE_PLUGIN=1 WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(PapyrusConsole PRIVATE /W4 /permissive- /Zc:preprocessor /EHsc /utf-8)
  target_link_options(PapyrusConsole PRIVATE "/PDBALTPATH:%_PDB%")
  add_library(native_analysis OBJECT EXCLUDE_FROM_ALL src/plugin.cpp src/Menu.cpp src/VM.cpp src/GetterDispatch.cpp src/Compiler.cpp src/VMImports.cpp)
  target_link_libraries(native_analysis PRIVATE CommonLibSSE::CommonLibSSE)
  target_include_directories(native_analysis PRIVATE src "${SKSE_MCP_INCLUDE_DIRS}")
  target_compile_definitions(native_analysis PRIVATE __CMAKE_COMMONLIBSSE_PLUGIN=1 WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(native_analysis PRIVATE /analyze /analyze:external- /WX /W4 /permissive- /Zc:preprocessor /EHsc /utf-8 "/FI${PROJECT_SOURCE_DIR}/src/PCH.h")
  get_target_property(COMMONLIB_INCLUDES CommonLibSSE INTERFACE_INCLUDE_DIRECTORIES)
  target_include_directories(native_analysis SYSTEM PRIVATE ${COMMONLIB_INCLUDES})
  install(TARGETS PapyrusConsole RUNTIME DESTINATION SKSE/Plugins)
endif()

enable_testing()
find_package(doctest CONFIG REQUIRED)
add_executable(console_tests tests/console_tests.cpp)
target_include_directories(console_tests PRIVATE src)
target_link_libraries(console_tests PRIVATE doctest::doctest)
if(MSVC)
  target_compile_options(console_tests PRIVATE /W4 /WX /utf-8)
endif()
add_test(NAME console_tests COMMAND console_tests)
if(WIN32)
  add_executable(compiler_tests tests/compiler_tests.cpp src/Compiler.cpp)
  target_include_directories(compiler_tests PRIVATE src)
  target_link_libraries(compiler_tests PRIVATE doctest::doctest bcrypt)
  target_compile_definitions(compiler_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(compiler_tests PRIVATE /W4 /WX /utf-8)
  add_test(NAME compiler_tests COMMAND compiler_tests)
endif()
